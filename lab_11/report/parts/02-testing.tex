\section{Тестирование}
Вопросы к системе представлены в листингах~\ref{lst:test_len}--\ref{lst:test_merge}.

\begin{listing}[Тестирование отношения len]{lst:test_len}{Prolog}
?- len([], L).
L = 0.

?- len([a], L).
L = 1.

?- len([a, 6, 9, d, 3], L).
L = 5.
\end{listing}

\begin{listing}[Тестирование отношения sum]{lst:test_sum}{Prolog}
?- sum([], S).
S = 0.

?- sum([52], S).
S = 52.

?- sum([10, -5, 20, -3], S).
S = 22.
\end{listing}

\begin{listing}[Тестирование отношения sum\_odd\_pos]{lst:test_sum_odd}{Prolog}
?- sum_odd_pos([10], S).
S = 0.

?- sum_odd_pos([10, 20, 30, 40], S).
S = 60.

?- sum_odd_pos([10, 20, 30, 40, 50], S).
S = 60.
\end{listing}

\begin{listing}[Тестирование отношения filter\_gt]{lst:test_filter}{Prolog}
?- filter_gt([2, 5, 8], 10, Result).
Result = [].

?- filter_gt([10, 20, 30], 5, Result).
Result = [30, 20, 10].

?- filter_gt([2, 8, 3, 10, 5], 5, Result).
Result = [10, 8].
\end{listing}

\begin{listing}[Тестирование отношения remove]{lst:test_remove}{Prolog}
?- remove([1, 2, 3], 5, Result).
Result = [3, 2, 1].

?- remove([3, 3, 3], 3, Result).
Result = [].

?- remove([1, 3, 2, 3, 4], 3, Result).
Result = [4, 2, 1].
\end{listing}

\begin{listing}[Тестирование отношения merge]{lst:test_merge}{Prolog}
?- merge([], [3, 4], Result).
Result = [3, 4].

?- merge([1, 2], [], Result).
Result = [2, 1].

?- merge([3, 1], [3, 7], Result).
Result = [1, 3, 3, 7].
\end{listing}
